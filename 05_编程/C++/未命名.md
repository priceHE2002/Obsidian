```c++
#include<iostream>
usint namespace std;
const int n = 10;

typedef struct MyQueue
{
	int* A;
	int front, rear;
}MyQueue;

void InitMyQueue(MyQueue &q)
{
	q.A = new int[n];
	q.front = 0;
	q.rear = n - 1;
}

bool enQueue(MyQueue &q, int data)
{
	if(q.front == (q.rear + 2) % n)
	{
		cout<<"队列已满，入队失败"<<endl;
		return false;
	}
	//若队列没满，则入队
	q.rear = (q.rear + 1) % n;
	q.A[q.rear] = data;
	return true;
}

bool deQueue(MyQueue &q, int &data)
{
	if(q.front == (q.rear + 1) % n)
	{
		cout<<"队列为空，出队失败"<<endl;
		return false;
	}
	//若队列不为空，则出队
	data = q.A[q.front];
	q.front = (q.front + 1) % n;
	return true;
}
```
