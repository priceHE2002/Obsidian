function match(p) {
  if (F.mid !== '全部' && p.母题 !== F.mid) return false;
  if (F.status !== '全部' && p.状态 !== F.status) return false;
  if (F.priority !== '全部' && p.优先级 !== F.priority) return false;
  if (F.source !== '全部' && p.来源 !== F.source) return false;
  return true;
}

function render() {
  const list = all.filter(match);
  const priRank = { 'S+':4, 'S':3, 'A':2, 'B':1 };
  const stRank = { '可纸笔':5, '已二刷':4, '已AC':3, '学习中':2, '未开始':1 };
  if (sortKey === '题号') list.sort((a,b) => sortAsc ? a.题号 - b.题号 : b.题号 - a.题号);
  else if (sortKey === '优先级') list.sort((a,b) => sortAsc ? (priRank[b.优先级]??0) - (priRank[a.优先级]??0) : (priRank[a.优先级]??0) - (priRank[b.优先级]??0));
  else if (sortKey === '状态') list.sort((a,b) => sortAsc ? (stRank[b.状态]??0) - (stRank[a.状态]??0) : (stRank[a.状态]??0) - (stRank[b.状态]??0));
  else if (sortKey === '母题') list.sort((a,b) => sortAsc ? a.母题.localeCompare(b.母题) : b.母题.localeCompare(a.母题));

  const chip = (opts, key) => opts.map(v =>
    `<span class="dst-chip${F[key] === v ? ' on' : ''}" data-filter="${key}|${v}">${v}</span>`).join('');

  const rows = list.map(p => {
    const path = p.file.path;
    const dots = statuses.map(s =>
      `<button class="dst-dot${p.状态 === s ? ' cur' : ''}" style="background:${stColor[s]};color:${stColor[s]}" title="切换为「${s}」" data-path="${path}" data-status="${s}"></button>`).join('');
    const link = `<a class="internal-link" href="${path}">${p.标题}</a>`;
    const d = (rateMap[p.状态] ?? 0) * 100;
    const priCls = p.优先级 === 'S+' ? 'dst-pri-sp' : 'dst-pri-' + p.优先级;
    const srcTag = p.来源 === '408真题' ? '<span class="dst-tag exam">真题</span>' : '<span class="dst-tag lc">LC</span>';
    const wrTag = p.手写 === '必须' ? '<span class="dst-tag must">必须</span>' : (p.手写 === '建议' ? '<span class="dst-tag" style="background:rgba(52,211,153,.14);color:#34d399;">建议</span>' : (p.手写 ? `<span class="dst-tag no">${p.手写}</span>` : '—'));
    return `<tr>
      <td class="dst-seq">${p.题号}</td>
      <td style="font-size:12px;color:var(--text-muted);font-weight:600;">${p.母题}</td>
      <td style="font-size:12px;color:var(--text-muted);font-variant-numeric:tabular-nums;">${p.引用}</td>
      <td>${link}</td>
      <td><span class="dst-pri ${priCls}">${p.优先级}</span></td>
      <td>${srcTag}</td>
      <td>${wrTag}</td>
      <td><span class="dst-st"><i style="background:${stColor[p.状态]}"></i>${p.状态}</span></td>
      <td><span class="dst-dots">${dots}</span></td>
      <td><span class="dst-bar"><i style="width:${d}%;background:${stColor[p.状态]};"></i></span><span class="dst-pct">${d.toFixed(0)}%</span></td>
    </tr>`;
  }).join('');

  const n = list.length;
  const nDone = list.filter(p => (rateMap[p.状态] ?? 0) >= 0.5).length;
  const sumRate = n ? list.reduce((a, p) => a + (rateMap[p.状态] ?? 0), 0) / n : 0;
  const mids = ['全部', ...new Set(all.map(p => p.母题))];

  content.innerHTML = `
    <div class="dst-filters">
      <div class="dst-frow"><span class="dst-flab">母题</span>${chip(mids, 'mid')}</div>
      <div class="dst-frow"><span class="dst-flab">状态</span>${chip(['全部', ...statuses], 'status')}</div>
      <div class="dst-frow"><span class="dst-flab">优先级</span>${chip(['全部','S+','S','A','B'], 'priority')}</div>
      <div class="dst-frow"><span class="dst-flab">来源</span>${chip(['全部','LeetCode','408真题'], 'source')}</div>
    </div>
    <div class="dst-summary"><span>筛选出 <b>${n}</b> 题 · 已完成(AC以上) <b>${nDone}</b> · 组内完成率 <b>${(sumRate*100).toFixed(0)}%</b></span><span class="dst-hint">点击色块切换状态 · 点击表头排序</span></div>
    <table class="dst-table">
      <thead><tr>
        <th data-sort="题号"># ${sortKey==='题号'?(sortAsc?'▲':'▼'):''}</th>
        <th data-sort="母题">母题 ${sortKey==='母题'?(sortAsc?'▲':'▼'):''}</th>
        <th>题号/年份</th>
        <th>题目</th>
        <th data-sort="优先级">优先级 ${sortKey==='优先级'?(sortAsc?'▲':'▼'):''}</th>
        <th>来源</th>
        <th>手写</th>
        <th data-sort="状态">状态 ${sortKey==='状态'?(sortAsc?'▲':'▼'):''}</th>
        <th>快速切换</th>
        <th>完成率</th>
      </tr></thead>
      <tbody>${rows || '<tr><td colspan="10" style="text-align:center;color:var(--text-muted);padding:18px;">无匹配题目</td></tr>'}</tbody>
    </table>`;
}

root.addEventListener('click', async (e) => {
  const dot = e.target.closest('button[data-status]');
  if (dot) {
    const path = dot.dataset.path, status = dot.dataset.status;
    const file = dv.app.vault.getAbstractFileByPath(path);
    if (file) {
      await dv.app.fileManager.processFrontMatter(file, fm => { fm['状态'] = status; });
      // 立即更新本地数据并重绘表格，避免等待 Dataview 索引刷新导致"点两次才生效"
      const rec = all.find(x => x.file.path === path);
      if (rec) rec.状态 = status;
      render();
      // 其他统计视图由 Dataview 的 metadata-change 事件自动刷新
    }
    return;
  }
  const chip = e.target.closest('span[data-filter]');
  if (chip) {
    const [k, v] = chip.dataset.filter.split('|');
    F[k] = v;
    render();
    return;
  }
  const th = e.target.closest('th[data-sort]');
  if (th) {
    const k = th.dataset.sort;
    if (sortKey === k) sortAsc = !sortAsc; else { sortKey = k; sortAsc = true; }
    render();
  }
});

render();
