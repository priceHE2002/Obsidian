const root = dv.el('div');
root.createEl('style', { text: CSS });
const view = root.createEl('div');

// ============ 数据模型 ============
const CATS = ['零壹每日一题','27代码打卡营','27数据结构强化','27组成原理强化','27操作系统强化','27计网强化'];
const ST = [['未完成','#9aa4b2'],['已完成','#34d399']];
const today = dv.date('today');
const todayStr = today.toFormat('yyyy-MM-dd');
const weekAgoStr = today.minus({ days: 7 }).toFormat('yyyy-MM-dd');

const pages = dv.pages('"12_考研打卡"').where(p => p.序号 && p.课程);
const all = pages.sort(p => (p.课程序 || 0) * 1000 + (p.序号 || 0), 'asc').values;

const esc = s => String(s ?? '').replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;');
// 完成日期统一转 yyyy-MM-dd（Dataview 会把 frontmatter 日期解析成 Luxon DateTime）
const fmtDate = v => {
  if (!v) return '';
  if (v.toFormat) return v.toFormat('yyyy-MM-dd');
  return String(v).slice(0, 10);
};

// ============ 交互状态 ============
let catFilter = '全部';
let statusFilter = '全部';
let sortKey = '序';
let sortAsc = true;

function sortVal(p) {
  switch (sortKey) {
    case '序': return (p.课程序 || 0) * 1000 + (p.序号 || 0);
    case '课程': return p.课程 || '';
    case '标题': return p.标题 || '';
    case '时长': { const m = /(\d+):(\d+)/.exec(p.时长 || ''); return m ? (+m[1] * 60 + +m[2]) : 0; }
    case '状态': return p.状态 === '已完成' ? 1 : 0;
    case '日期': return fmtDate(p.完成日期);
    default: return 0;
  }
}

// ============ 渲染 ============
function render() {
  const total = all.length;
  const counts = {};
  ST.forEach(([k]) => { counts[k] = all.filter(p => p.状态 === k).length; });
  const done = counts['已完成'];
  const pct = total ? Math.round(done / total * 100) : 0;
  const doneToday = all.filter(p => fmtDate(p.完成日期) === todayStr).length;
  const doneWeek = all.filter(p => { const d = fmtDate(p.完成日期); return d && d >= weekAgoStr && d <= todayStr; }).length;

  // —— 签名元素：多段进度环 ——
  const segs = [];
  let acc = 0;
  for (const [k, c] of ST) {
    const deg = total ? counts[k] / total * 360 : 0;
    if (deg > 0) { segs.push(`${c} ${acc.toFixed(2)}deg ${(acc + deg).toFixed(2)}deg`); acc += deg; }
  }
  if (!segs.length) segs.push('#3a3f47 0deg 360deg');

  const ringHTML = `
  <div style="position:relative;width:168px;height:168px;border-radius:50%;
    background:conic-gradient(${segs.join(',')});
    filter:drop-shadow(0 6px 18px rgba(52,211,153,.18));">
    <div style="position:absolute;inset:13px;border-radius:50%;background:var(--background-primary);
      box-shadow:inset 0 1px 3px rgba(0,0,0,.06);display:flex;flex-direction:column;align-items:center;justify-content:center;">
      <b style="font-size:38px;font-weight:800;letter-spacing:-.02em;font-variant-numeric:tabular-nums;
        background:linear-gradient(135deg,var(--text-normal),var(--interactive-accent));-webkit-background-clip:text;
        background-clip:text;-webkit-text-fill-color:transparent;">${pct}%</b>
      <span style="font-size:10px;letter-spacing:.14em;text-transform:uppercase;color:var(--text-muted);">完成率</span>
      <span style="font-size:11px;color:var(--text-muted);margin-top:2px;">已完成 ${done} / ${total}</span>
    </div>
  </div>
  <div style="margin-left:18px;font-size:12.5px;color:var(--text-muted);line-height:1.9;">
    ${ST.map(([k, c]) => `<div><i style="width:11px;height:11px;border-radius:3px;background:${c};display:inline-block;margin-right:8px;"></i>${k} · <b style="color:var(--text-normal);">${counts[k]}</b></div>`).join('')}
  </div>`;

  // —— KPI 卡片 ——
  const kpis = [
    ['总课程', total], ['已完成', done], ['完成率', pct + '%'],
    ['剩余', total - done], ['今日打卡', doneToday], ['本周打卡', doneWeek]
  ];
  const kpiHTML = `<div class="dsb-kpis">` +
    kpis.map(([lab, val]) => `<div class="dsb-kpi"><span>${lab}</span><b>${val}</b></div>`).join('') + `</div>`;

  // —— 状态分布分段条 ——
  const distHTML = `<div class="dsb-bar">` +
    ST.map(([k, c]) => {
      const w = total ? (counts[k] / total * 100).toFixed(1) : 0;
      return `<i title="${k} ${counts[k]} 节" style="width:${w}%;background:${c};"></i>`;
    }).join('') + `</div>
  <div class="dsb-legend">
    ${ST.map(([k, c]) => `<span><i style="background:${c};"></i>${k}（${counts[k]}）</span>`).join('')}
  </div>`;

  // —— 专题进度 ——
  const catHTML = CATS.map(cat => {
    const items = all.filter(p => p.课程 === cat);
    const n = items.length;
    const d = items.filter(p => p.状态 === '已完成').length;
    const r = n ? d / n : 0;
    const color = r >= .75 ? '#fbbf24' : r >= .5 ? '#34d399' : r > 0 ? '#5b8cff' : 'var(--text-muted)';
    const pc = Math.round(r * 100);
    return `<div class="dsb-cat-row">
      <span class="dsb-cat-name" title="${esc(cat)}">${esc(cat)}</span>
      <span class="dsb-cat-track"><i class="dsb-cat-fill" style="width:${pc}%;background:linear-gradient(90deg,#5b8cff,${color});"></i></span>
      <span class="dsb-cat-pct" style="color:${color};">${pc}%</span>
      <span class="dsb-cat-n">${d}/${n}</span>
    </div>`;
  }).join('');

  // —— 交互表格 ——
  const chipRow = `
  <div style="margin-bottom:8px;">
    <div class="dsb-sec-title">专题筛选</div>
    ${['全部', ...CATS].map(c => `<span class="dst-chip ${catFilter === c ? 'on' : ''}" data-kind="cat" data-val="${esc(c)}">${esc(c)}</span>`).join('')}
  </div>
  <div style="margin-bottom:14px;">
    <div class="dsb-sec-title">状态筛选</div>
    ${['全部', '未完成', '已完成'].map(s => `<span class="dst-chip ${statusFilter === s ? 'on' : ''}" data-kind="status" data-val="${esc(s)}">${esc(s)}</span>`).join('')}
  </div>`;

  const cols = [['序','序'], ['课程','课程'], ['标题','标题'], ['时长','时长'], ['状态','状态'], ['日期','日期']];
  const thHTML = `<tr>` + cols.map(([k, lab]) => {
    const arrow = sortKey === k ? (sortAsc ? ' ▲' : ' ▼') : '';
    return `<th data-sort="${k}">${lab}${arrow}</th>`;
  }).join('') + `</tr>`;

  let rows = all.filter(p =>
    (catFilter === '全部' || p.课程 === catFilter) &&
    (statusFilter === '全部' || p.状态 === statusFilter));
  rows = rows.slice().sort((a, b) => {
    const va = sortVal(a), vb = sortVal(b);
    if (va < vb) return sortAsc ? -1 : 1;
    if (va > vb) return sortAsc ? 1 : -1;
    return 0;
  });

  const rowsHTML = rows.length
    ? rows.map(p => {
        const cur = p.状态;
        return `<tr>
          <td style="color:var(--text-muted);font-variant-numeric:tabular-nums;">${p.序号}</td>
          <td style="color:var(--text-muted);white-space:nowrap;">${esc(p.课程)}</td>
          <td><span class="dsb-open" data-open="${esc(p.file.path)}" title="点击打开笔记">${esc(p.标题)}</span></td>
          <td style="color:var(--text-muted);font-variant-numeric:tabular-nums;">${esc(p.时长)}</td>
          <td style="white-space:nowrap;">${ST.map(([k, c]) =>
            `<button class="dst-dot ${cur === k ? 'cur' : ''}" data-path="${esc(p.file.path)}" data-status="${esc(k)}" style="background:${c};" title="标记为${k}"></button>`).join('')}
          </td>
          <td style="color:var(--text-muted);white-space:nowrap;">${fmtDate(p.完成日期)}</td>
        </tr>`;
      }).join('')
    : `<tr><td colspan="6"><div class="dsb-empty">🎉 当前筛选下没有课程</div></td></tr>`;

  const tableHTML = `
  <div class="dsb-sec-title">课程清单（点击状态点打卡 / 取消，点击表头排序）</div>
  <div style="margin-bottom:6px;" class="dsb-muted">当前显示 <b style="color:var(--text-normal);">${rows.length}</b> 节课</div>
  <table class="dst-table"><thead>${thHTML}</thead><tbody>${rowsHTML}</tbody></table>`;

  view.innerHTML = `
  <div class="dsb">
    <div class="dsb-hero">
      <div class="dsb-card dsb-ring-card">${ringHTML}</div>
      <div class="dsb-card dsb-kpi-card">${kpiHTML}</div>
    </div>
    <div class="dsb-stack">
      <div class="dsb-card"><div class="dsb-sec-title">完成状态分布</div>${distHTML}</div>
      <div class="dsb-card"><div class="dsb-sec-title">各专题进度</div>${catHTML}</div>
      <div class="dsb-card">${chipRow}${tableHTML}</div>
    </div>
  </div>`;
}

// ============ 事件委托 ============
root.addEventListener('click', async e => {
  const dot = e.target.closest('button.dst-dot');
  if (dot) {
    const path = dot.dataset.path;
    const status = dot.dataset.status;
    const file = dv.app.vault.getAbstractFileByPath(path);
    if (file) {
      await dv.app.fileManager.processFrontMatter(file, fm => {
        fm['状态'] = status;
        if (status === '已完成') fm['完成日期'] = todayStr;
        else delete fm['完成日期'];
      });
      const rec = all.find(x => x.file.path === path);
      if (rec) {
        rec.状态 = status;
        rec['完成日期'] = (status === '已完成') ? todayStr : '';
      }
      render();
    }
    return;
  }
  // 打开笔记（用 Obsidian 官方 openLinkText API，行为与点击内部链接一致）
  const opener = e.target.closest('.dsb-open');
  if (opener) {
    const path = opener.dataset.open || '';
    const linktext = path.startsWith('12_考研打卡/') ? path.slice('12_考研打卡/'.length).replace(/\.md$/, '') : path.replace(/\.md$/, '');
    await dv.app.workspace.openLinkText(linktext, dv.current().file.path, e.ctrlKey || e.metaKey);
    return;
  }
  const chip = e.target.closest('.dst-chip');
  if (chip) {
    if (chip.dataset.kind === 'cat') catFilter = chip.dataset.val;
    else statusFilter = chip.dataset.val;
    render();
    return;
  }
  const th = e.target.closest('th[data-sort]');
  if (th) {
    const k = th.dataset.sort;
    if (sortKey === k) sortAsc = !sortAsc;
    else { sortKey = k; sortAsc = true; }
    render();
    return;
  }
});

render();
