root.addEventListener('click', async e => {
  const btn = e.target.closest('button[data-s]'); if(!btn) return;
  const s = btn.dataset.s;
  const tf = dv.app.vault.getAbstractFileByPath(cur.file.path);
  await dv.app.fileManager.processFrontMatter(tf, fm => { fm['状态'] = s; fm['完成日期'] = (s==='已完成') ? dv.date('today').toFormat('yyyy-MM-dd') : ''; });
