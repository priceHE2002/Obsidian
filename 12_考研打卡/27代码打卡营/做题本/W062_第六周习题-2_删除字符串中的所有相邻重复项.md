const cur = dv.current();
const statuses = ['未完成','已完成'];
const colors = {'未完成':'#9aa4b2','已完成':'#34d399'};
const root = dv.el('div','');
const render = () => {
  root.innerHTML = '<div style="display:flex;gap:8px;flex-wrap:wrap;">' + statuses.map(s =>
    '<button data-s="'+s+'" style="padding:5px 16px;font-size:13px;border-radius:16px;border:1px solid var(--background-modifier-border);cursor:pointer;background:var(--background-primary);color:var(--text-normal);' + (cur.状态===s ?   ).join('') + '</div>';
};
render();
root.addEventListener('click', async e => {
  const btn = e.target.closest('button[data-s]'); if(!btn) return;
  const s = btn.dataset.s;
  const tf = dv.app.vault.getAbstractFileByPath(cur.file.path);
  await dv.app.fileManager.processFrontMatter(tf, fm => { fm['状态'] = s; fm['完成日期'] = (s==='已完成') ? dv.date('today').toFormat('yyyy-MM-dd') : ''; });
  cur.状态 = s;
  render();
});
