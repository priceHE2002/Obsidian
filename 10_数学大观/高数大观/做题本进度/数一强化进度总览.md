# 数一强化 · 进度总览

> [!abstract] 使用提示
> 在下方**交互表格**中点状态圆点即可打卡；或进入某道题笔记 / 某模块笔记单独打卡。所有进度实时汇总。
> 每题含题源（真题年份 / 660 / 880 / 900 / 1000 / 姜晓千真题同源）与自动提取的题干预览。
> 详细口径见 [[使用说明]]。

const F = dv.current().file.folder;

const state = { chap:null, cat:null, stat:null, key:null, review:false, sort:['序','asc'], kw:'' };

function applyFilter(){
  filtered = all.filter(p => {
    if (state.chap && p.章 !== state.chap) return false;
    if (state.cat && p['题源大类'] !== state.cat) return false;
    if (state.stat && p.状态 !== state.stat) return false;
    if (state.key && !p.重点) return false;
    if (state.review && (!p['下次复习'] || p.状态 === '已掌握')) return false;
    if (state.kw && !(String(p['源题号']).includes(state.kw) || String(p['模块名']).includes(state.kw) || String(p.题源||'').includes(state.kw))) return false;
    return true;
  });
  const [key, dir] = state.sort;
  const rank = (a,b) => {
    if (key === '序' || key === '页码') return dir==='asc' ? a-key - b-key : b-key - a-key;
    if (key === '模块'){
      const pa = String(a['模块']).split('.').map(Number), pb = String(b['模块']).split('.').map(Number);
      for (let i=0;i<Math.max(pa.length,pb.length);i++){ const x = pa[i]||0, y = pb[i]||0; if (x!==y) return dir==='asc'? x-y : y-x; }
      return 0;
    }
    const va = String(a[key] ?? ''), vb = String(b[key] ?? '');
    return dir==='asc' ? va.localeCompare(vb, 'zh') : vb.localeCompare(va, 'zh');
  };
  filtered.sort(rank);
}
